### Reference file

@startuml
actor "Registered User" as User
participant "System" as Sys

activate User


== Enable automated mode ==
User -> User: click_on(toggle_automated_mode)

User -> Sys: request_enable_automated_mode()
activate Sys

Sys -> Sys: check_permissions(gps, accelerometer, gyroscope, background_execution)
alt permissions_not_granted (E1)
    Sys -> User: show(prompt_request_permissions)
    User --> Sys: grant | deny
    alt user_grants_permissions    
        Sys -> Sys: check_sensors()
        alt gps_or_sensor_error (E2)
            Sys --> User: show(error_sensor)
        else else
            Sys -> Sys: activate(automated_mode)
            Sys --> User: notify(automatic_mode_enabled)
        end
    else else
        Sys --> User: show(error_permissions_denied)
    end
else else
    Sys -> Sys: check_sensors()
    alt gps_or_sensor_error (E2)
        Sys --> User: show(error_sensor)
    else else
        Sys -> Sys: activate(automated_mode)
        Sys --> User: notify(automatic_mode_enabled)
    end
end

deactivate Sys
deactivate User

@enduml
