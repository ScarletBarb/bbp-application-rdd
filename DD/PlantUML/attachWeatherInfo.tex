participant TripDB as TDB
participant WeatherAPI as W

loop for_each_job
TS -> W: ping()
activate W
alt api_not_available
TS -> TS: scheduleJobLater(trip_id)
else else 
W --> TS: PONG
TS -> W: getWeather(trip.timestamp, trip.location)
alt weather_data_not_available
W --> TS: NO_DATA
TS -> TDB: uptadeTrip(trip_id, weather_data_not_available)
activate TDB

else else

activate TDB
TDB --> TS: OK
deactivate TDB
end
end
end
